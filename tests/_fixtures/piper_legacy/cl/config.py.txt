class Cfg(dict):
    """Dict with attribute access, so call sites read like prose."""

    def __getattr__(self, name: str) -> Any:
        try:
            value = self[name]
        except KeyError as error:
            raise AttributeError(name) from error
        return Cfg(value) if isinstance(value, dict) else value

    def get_path(self, dotted: str, default: Any = None) -> Any:
        node: Any = self
        for part in dotted.split("."):
            if not isinstance(node, dict) or part not in node:
                return default
            node = node[part]
        return node


def _merge(base: dict, override: dict) -> dict:
    out = dict(base)
    for key, value in override.items():
        if isinstance(value, dict) and isinstance(out.get(key), dict):
            out[key] = _merge(out[key], value)
        else:
            out[key] = value
    return out


def load_profile() -> Cfg:
    DATA.mkdir(parents=True, exist_ok=True)
    if PROFILE.exists():
        try:
            override = json.loads(PROFILE.read_text(encoding="utf-8"))
        except (OSError, json.JSONDecodeError):
            override = {}
        return Cfg(_merge(DEFAULTS, override))
    return Cfg(_merge(DEFAULTS, {}))


def write_profile(patch: dict) -> Cfg:
    current = json.loads(PROFILE.read_text(encoding="utf-8")) if PROFILE.exists() else {}
    DATA.mkdir(parents=True, exist_ok=True)
    PROFILE.write_text(json.dumps(_merge(current, patch), indent=2) + "\n", encoding="utf-8")
    return load_profile()


def load_json(path: Path) -> dict:
    return json.loads(path.read_text(encoding="utf-8"))


def dump_json(path: Path, payload: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")


def handeye() -> dict:
    return load_json(HANDEYE)
